// Same fake lifecycle as pool-check.ts, with a plain-record representation.
interface FakeSandbox extends Sandbox {reap:()=>void;kills:number}
interface Fake extends Backend {activeStarts:number;maxStarts:number;instances:FakeSandbox[]}
function fake(autoReap=true):Fake {
 const b:Fake={kind:'local-qjs',memoryLimitMB:32,activeStarts:0,maxStarts:0,instances:[],
  park:async(signal?:AbortSignal):Promise<Sandbox>=>{
   b.activeStarts++;b.maxStarts=Math.max(b.maxStarts,b.activeStarts);
   let reap!:()=>void;const exited=new Promise<void>(r=>{reap=r;});
   const s:FakeSandbox={exited,reap,kills:0,read:async():Promise<unknown>=>null,send:async(_frame:unknown):Promise<void>=>{},kill:()=>{s.kills++;if(autoReap)setTimeout(reap,20);}};
   b.instances.push(s);
   try{await delay(10);if(signal?.aborted){s.reap();throw Error('aborted');}return s;}finally{b.activeStarts--;}
  }};
 return b;
}
